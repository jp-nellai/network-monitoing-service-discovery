cmake_minimum_required(VERSION 3.16)
project(nms_discovery_v2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(SQLite3 REQUIRED)
find_path(NETSNMP_INCLUDE_DIR net-snmp/net-snmp-config.h)
find_library(NETSNMP_LIBRARY netsnmp)
if(NOT NETSNMP_INCLUDE_DIR OR NOT NETSNMP_LIBRARY)
  message(FATAL_ERROR "Net-SNMP development files not found")
endif()
add_executable(nms-discovery src/main.cpp src/snmp_client.cpp src/discovery.cpp src/network_scanner.cpp src/database.cpp src/http_server.cpp src/vendor.cpp)
target_include_directories(nms-discovery PRIVATE include ${NETSNMP_INCLUDE_DIR})
target_link_libraries(nms-discovery PRIVATE ${NETSNMP_LIBRARY} SQLite::SQLite3 Threads::Threads)
if(NOT MSVC)
 target_compile_options(nms-discovery PRIVATE -Wall -Wextra -Wpedantic)
endif()
